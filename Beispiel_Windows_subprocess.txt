import subprocess

process = subprocess.run (["dir"], check=True, \
stdout=subprocess.PIPE, stderr=subprocess.PIPE, universal_newlines=True,shell=True)
    
output = process.stdout.encode("cp850", errors="replace")
error_output = process.stderr.encode("cp850", errors="replace")
print("Positiv", output)
print("Fehler", error_output)
